function captureDefinition(): { description: string } {
  const captured = new Map<string, { description: string }>()
  const ctx = {
    get(name: string): unknown {
      if (name !== "tools") return undefined
      return {
        register(definition: { name: string } & Record<string, unknown>): () => void {
          captured.set(definition.name, definition as never)
          return () => undefined
        },
      }
    },
  }
  registerWfExperienceFeedback(ctx, {} as never)
  const definition = captured.get("wf_experience_feedback")
  if (!definition) throw new Error("工具未注册")
  return definition
}

async function readSource(relative: string): Promise<string> {
  return readFile(fileURLToPath(new URL(`../../src/${relative}`, import.meta.url)), "utf8")
}

async function collectSources(): Promise<Array<{ path: string; text: string }>> {
  const root = fileURLToPath(new URL("../../src/", import.meta.url))
  const collected: Array<{ path: string; text: string }> = []
  const walk = async (dir: string): Promise<void> => {
    for (const entry of await readdir(dir, { withFileTypes: true })) {
      const full = join(dir, entry.name)
      if (entry.isDirectory()) {
        await walk(full)
        continue
      }
      if (!entry.name.endsWith(".ts") && !entry.name.endsWith(".tsx")) continue
      collected.push({ path: full.replace(root, "").replace(/\\/gu, "/"), text: await readFile(full, "utf8") })
    }
  }
  await walk(root)
  return collected
}

/**
 * 我对方案 §5 的逐条判据：每一级必须命中的语义关键词（取自 §5 中文定义的语义要点）。
 * 任一关键词缺失即视为该级语义在运行期文本中被改掉或弱化到不可判定。
 */
const SECTION5_MARKERS: ReadonlyArray<{ dimension: string; value: number; markers: readonly string[] }> = [
  // §5.1 Fit：0 基本不适用/关键条件不成立；0.25 只有局部关联/关键适用条件不足；
  //          0.5 部分适用/需要明显调整后才能使用；0.75 大部分关键条件成立/可直接参考；1 高度匹配/可直接迁移判断
  { dimension: "fitScore", value: 0, markers: ["key conditions do not hold", "not applicable"] },
  { dimension: "fitScore", value: 0.25, markers: ["locally related", "conditions missing"] },
  { dimension: "fitScore", value: 0.5, markers: ["partially applicable", "adjustments"] },
  { dimension: "fitScore", value: 0.75, markers: ["most key conditions hold", "directly"] },
  { dimension: "fitScore", value: 1, markers: ["closely matches", "transfers directly"] },
  // §5.2 Decision Effect：-1 错误方向/重大风险返工损失；-0.5 明显负面影响/未形成严重后果；
  //          0 无归因差异；+0.5 明显正面帮助/降低风险成本返工或提高质量；+1 决定性帮助/没有它决策很可能明显变差
  { dimension: "decisionEffect", value: -1, markers: ["wrong way", "risk", "rework", "loss"] },
  { dimension: "decisionEffect", value: -0.5, markers: ["harmed", "without severe consequences"] },
  { dimension: "decisionEffect", value: 0, markers: ["no attributable difference"] },
  { dimension: "decisionEffect", value: 0.5, markers: ["helped", "less risk", "better result"] },
  { dimension: "decisionEffect", value: 1, markers: ["decisive", "key outcome", "worse"] },
  // §5.3 Information Gain：0 几乎是常识/空泛正确话术/不提供额外决策信息；0.25 很小的补充信息；
  //          0.5 有用但不强的区分条件；0.75 明显改变决策边界或行动选择；1 关键非显然强区分性决策规则
  { dimension: "informationGain", value: 0, markers: ["common sense", "no extra decision information"] },
  { dimension: "informationGain", value: 0.25, markers: ["very small supplementary"] },
  { dimension: "informationGain", value: 0.5, markers: ["useful but weak", "discriminating condition"] },
  { dimension: "informationGain", value: 0.75, markers: ["decision boundary", "chosen action"] },
  { dimension: "informationGain", value: 1, markers: ["key", "non-obvious", "discriminating decision rule"] },
  // §5.4 Causal Confidence：0 无法判断是否由经验导致；0.25 非常弱的主观猜测；
  //          0.5 有一定证据但存在明显混杂因素；0.75 较强证据表明经验实际影响了决策；1 直接归因/明确反事实证据
  { dimension: "causalConfidence", value: 0, markers: ["cannot tell whether the experience caused it"] },
  { dimension: "causalConfidence", value: 0.25, markers: ["very weak subjective guess"] },
  { dimension: "causalConfidence", value: 0.5, markers: ["some evidence", "confounding factors"] },
  { dimension: "causalConfidence", value: 0.75, markers: ["strong evidence", "influenced the decision"] },
  { dimension: "causalConfidence", value: 1, markers: ["direct attribution", "counterfactual"] },
]

const SECTION5_QUESTIONS: ReadonlyArray<{ dimension: string; markers: readonly string[] }> = [
  { dimension: "fitScore", markers: ["apply to the current situation"] },
  { dimension: "decisionEffect", markers: ["compared with not using it", "actually change"] },
  { dimension: "informationGain", markers: ["discriminating decision information", "provide here"] },
  { dimension: "causalConfidence", markers: ["how confident", "own actions, external systems or chance"] },
]

function anchorOf(dimension: string, value: number): string {
  const found = EVALUATION_DIMENSION_DEFINITIONS.find((entry) => entry.dimension === dimension)
  if (!found) throw new Error(`缺维度 ${dimension}`)
  const anchor = found.anchors.find((entry) => entry.value === value)
  if (!anchor) throw new Error(`缺锚点 ${dimension}@${value}`)
  return anchor.definition.toLowerCase()
}

describe("第三轮复核：压缩后的锚点表", () => {
  it("test_注册后description_仍20条完整注入且各出现一次_长度已下降", () => {
    const definition = captureDefinition()
    let checked = 0
    for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
      expect(definition.description).toContain(dimension.parameter)
      for (const anchor of dimension.anchors) {
        const entry = `${anchor.value} = ${anchor.definition}`
        expect(definition.description, `缺 ${dimension.dimension}@${anchor.value}`).toContain(entry)
        expect(definition.description.split(entry).length - 1, `重复 ${entry}`).toBe(1)
        checked += 1
      }
    }
    expect(checked).toBe(20)
    expect(definition.description).toContain(renderAnchorGlossary())
    const glossaryShare = renderAnchorGlossary().length / definition.description.length
    // 断言与测试护栏一致：护栏上限 3400，且确有可观余量
    expect(definition.description.length).toBeLessThan(3400)
    expect(definition.description.length).toBeGreaterThan(2000)
    // eslint-disable-next-line no-console
    console.log(`\n第三轮证据：description=${definition.description.length}（护栏 3400，余量 ${(3400 - definition.description.length)}）glossary=${renderAnchorGlossary().length}（占 ${(glossaryShare * 100).toFixed(1)}%）\n`)
  })

  it("test_20条定义与4个问题_逐条命中方案§5语义要点", () => {
    for (const item of SECTION5_MARKERS) {
      const definition = anchorOf(item.dimension, item.value)
      for (const marker of item.markers) {
        expect(definition, `${item.dimension}@${item.value} 缺 §5 语义「${marker}」`).toContain(marker)
      }
    }
    for (const item of SECTION5_QUESTIONS) {
      const found = EVALUATION_DIMENSION_DEFINITIONS.find((entry) => entry.dimension === item.dimension)
      if (!found) throw new Error(`缺维度 ${item.dimension}`)
      for (const marker of item.markers) {
        expect(found.question.toLowerCase(), `${item.dimension} 问题缺 §5 语义「${marker}」`).toContain(marker)
      }
    }
    expect(SECTION5_MARKERS).toHaveLength(20)
    expect(SECTION5_QUESTIONS).toHaveLength(4)
  })

  it("test_20条定义仍互不重复且非相对措辞", () => {
    const all = EVALUATION_DIMENSION_DEFINITIONS.flatMap((d) => d.anchors.map((a) => a.definition))
    expect(all).toHaveLength(20)
    expect(new Set(all).size).toBe(20)
    for (const text of all) {
      expect(text.split(/\s+/u).length).toBeGreaterThanOrEqual(3)
      expect(text.toLowerCase()).not.toMatch(/^(lower|higher|medium|moderate|average)\.?$/u)
    }
  })

  it("test_定义文本在src全域仍只出现在本体一处", async () => {
    const sources = await collectSources()
    for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
      for (const anchor of dimension.anchors) {
        const holders = sources.filter((file) => file.text.includes(anchor.definition)).map((file) => file.path)
        expect(holders, `定义「${anchor.definition}」出现于 ${holders.join(",")}`).toEqual([
          "host/experience/anchor-definitions.ts",
        ])
      }
    }
  })
})

describe("第三轮复核：类型别名与依赖方向", () => {
  it("test_源码_维度类型是别名而非重写的联合_且import为type-only", async () => {
    const source = await readSource("host/experience/anchor-definitions.ts")
    expect(source).toContain("export type EvaluationDimension = EvaluationScoreField")
    expect(source).toContain('import type { EvaluationScoreField } from "./scoring.js"')
    // 不再出现重写的联合字面量
    expect(source).not.toMatch(/"fitScore"\s*\|/u)
    expect(source).not.toMatch(/=\s*"fitScore"/u)
  })

  it("test_依赖方向_scoring不反向依赖anchor_definitions_且无环", async () => {
    const scoring = await readSource("host/experience/scoring.ts")
    expect(scoring).not.toContain("anchor-definitions")
    const definitions = await readSource("host/experience/anchor-definitions.ts")
    // 唯一的运行时依赖是 constants（值）；scoring 只以 type 形态出现
    const runtimeImports = definitions
      .split(/\r?\n/u)
      .map((line) => line.trim())
      .filter((line) => /^import\b/u.test(line))
      .filter((line) => !line.startsWith("import type "))
    expect(runtimeImports).toEqual(['import { DECISION_EFFECT_ANCHORS, SCORE_ANCHORS } from "./constants.js"'])
  })

  it("test_编译产物_anchor_definitions无scoring运行时边", async () => {
    const artifact = await readFile(fileURLToPath(new URL("../../lib/experience/anchor-definitions.js", import.meta.url)), "utf8")
    expect(artifact).not.toContain("scoring")
    const imports = artifact.split(/\r?\n/u).filter((line) => /\bimport\b|\brequire\(/u.test(line))
    expect(imports).toEqual(['import { DECISION_EFFECT_ANCHORS, SCORE_ANCHORS } from "./constants.js";'])
  })
})
