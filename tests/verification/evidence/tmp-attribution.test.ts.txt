// 独立验证者临时文件（task-8）：§36 归因（causal confidence 对长期可信度的控制程度）与
// 评分错误消息的领域分立。验证完成后删除。

import { describe, expect, it } from "vitest"
import {
  aggregateExperienceStats,
  calculateEffectiveEffect,
  calculateEvaluationWeight,
  validateEvaluationScores,
} from "../../src/host/experience/index.js"
import type { ExperienceEvaluationScores } from "../../src/host/shared/asset-types.js"

const score = (fitScore: number, causalConfidence: number, decisionEffect: number, informationGain: number): ExperienceEvaluationScores =>
  ({ fitScore, causalConfidence, decisionEffect, informationGain }) as ExperienceEvaluationScores

describe("§36 归因：causal confidence 控制可信程度", () => {
  it("test_同样一次负向评价_归因置信度从1降到0_统计权重与信任影响同步缩小", () => {
    const confident = aggregateExperienceStats({ evaluations: [score(1, 1, -1, 0)], recalledCount: 1 })
    const guessing = aggregateExperienceStats({ evaluations: [score(1, 0, -1, 0)], recalledCount: 1 })

    // 权重：C=1 → 1；C=0 → 0.25（仍保留最小权重，不完全丢弃）
    expect(calculateEvaluationWeight(score(1, 1, -1, 0))).toBeCloseTo(1, 12)
    expect(calculateEvaluationWeight(score(1, 0, -1, 0))).toBeCloseTo(0.25, 12)
    expect(confident.effectiveSampleCount).toBeCloseTo(1, 12)
    expect(guessing.effectiveSampleCount).toBeCloseTo(0.25, 12)

    // 长期价值与信任：C=0 时负向影响明显更小（信任更接近中性 0.5）
    expect(confident.empiricalValue).toBeCloseTo(-0.25, 12)
    expect(guessing.empiricalValue).toBeCloseTo(-0.25 / 3.25, 12)
    expect(confident.trust).toBeCloseTo(0.486780901540767, 12)
    expect(guessing.trust).toBeCloseTo(0.49893499657802733, 12)
    expect(Math.abs(0.5 - guessing.trust)).toBeLessThan(Math.abs(0.5 - confident.trust))

    // harm_count 与 harm_severity **不受** causal confidence 影响（口径见 §13 原文）
    expect(confident.harmCount).toBe(1)
    expect(guessing.harmCount).toBe(1)
    expect(confident.harmRate).toBe(1)
    expect(guessing.harmRate).toBe(1)
    expect(confident.harmSeverity).toBe(1)
    expect(guessing.harmSeverity).toBe(1)

    // eslint-disable-next-line no-console
    console.log(`\n归因证据：C=1 trust=${confident.trust} severity=${confident.harmSeverity}；C=0 trust=${guessing.trust} severity=${guessing.harmSeverity}\n`)
  })

  it("test_运行事实不参与评分折算_负向效果与信息增益无关", () => {
    // effective_effect 只由 decisionEffect / informationGain 决定：负向不因信息增益低而减刑
    for (const informationGain of [0, 0.25, 0.5, 0.75, 1]) {
      expect(calculateEffectiveEffect(score(1, 1, -1, informationGain))).toBe(-1)
      expect(calculateEffectiveEffect(score(1, 1, -0.5, informationGain))).toBe(-0.5)
    }
  })
})

describe("§6.1 评分错误消息的领域分立", () => {
  it("test_决策效果传0.25_被拒且消息只列跨零锚点", () => {
    const result = validateEvaluationScores({ fitScore: 1, decisionEffect: 0.25, informationGain: 1, causalConfidence: 1 })
    expect(result.ok).toBe(false)
    if (result.ok) throw new Error("应当被拒绝")
    expect(result.field).toBe("decisionEffect")
    expect(result.reason).toContain("-1 / -0.5 / 0 / 0.5 / 1")
    expect(result.reason).not.toContain("0.25 / 0.5 / 0.75")
    expect(result.reason).toContain("0.25")
  })

  it("test_连续小数0.73_被拒且消息给出五级锚点与实际值", () => {
    const result = validateEvaluationScores({ fitScore: 0.73, decisionEffect: 1, informationGain: 1, causalConfidence: 1 })
    expect(result.ok).toBe(false)
    if (result.ok) throw new Error("应当被拒绝")
    expect(result.field).toBe("fitScore")
    expect(result.reason).toContain("0 / 0.25 / 0.5 / 0.75 / 1")
    expect(result.reason).toContain("0.73")
  })

  it("test_合法锚点_四维全部收窄通过", () => {
    const result = validateEvaluationScores({ fitScore: 0.25, decisionEffect: -0.5, informationGain: 0.75, causalConfidence: 0 })
    expect(result.ok).toBe(true)
    if (!result.ok) throw new Error("应当通过")
    expect(result.scores).toEqual({ fitScore: 0.25, decisionEffect: -0.5, informationGain: 0.75, causalConfidence: 0 })
  })

  it("test_字段缺失_给出同一套可行动原因", () => {
    const result = validateEvaluationScores({ fitScore: 1, decisionEffect: 1, informationGain: 1 })
    expect(result.ok).toBe(false)
    if (result.ok) throw new Error("应当被拒绝")
    expect(result.field).toBe("causalConfidence")
    expect(result.reason).toContain("undefined")
  })
})
