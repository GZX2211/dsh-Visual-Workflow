// 独立验证者临时文件（task-8）：
//   A. §36「Trust 只能在 semantic Top-K 内进行有界重排」——构造 35 条经验，其中 5 条落在
//      语义候选池之外却拥有高信任（其有界修正后的相关性高于池内任何一条），验证它们不得出现。
//   B. §24 增量路径：已有统计行时记录使用只涨 recalled_count，用**推进时钟**证明 updated_at 未被刷新。

import { afterEach, describe, expect, it } from "vitest"
import { mkdtemp, rm } from "node:fs/promises"
import { tmpdir } from "node:os"
import { join } from "node:path"
import { AssetStore } from "../../src/host/assets/index.js"
import {
  CANDIDATE_POOL_SIZE,
  ExperienceService,
  NEUTRAL_STATS,
  calculateRecallAdjustment,
  type ExperienceEmbeddingPort,
  type ExperienceRuntimePort,
  type ExperienceStorePort,
} from "../../src/host/experience/index.js"
import type { ExperienceStatsEntry } from "../../src/host/shared/asset-types.js"

const cleanups: Array<() => Promise<void>> = []
afterEach(async () => {
  await Promise.all(cleanups.splice(0).map((fn) => fn()))
})

const DIM = 40
/** 查询向量：e1。池内经验与查询余弦 0.5，池外经验 0.49。 */
const IN_COS = 0.5
const OUT_COS = 0.49

function inVector(index: number): Float64Array {
  const vector = new Float64Array(DIM)
  vector[0] = IN_COS
  vector[1 + index] = Math.sqrt(1 - IN_COS * IN_COS)
  return vector
}

function outVector(index: number): Float64Array {
  const vector = new Float64Array(DIM)
  vector[0] = OUT_COS
  vector[31 + index] = Math.sqrt(1 - OUT_COS * OUT_COS)
  return vector
}

const UNKNOWN = (() => {
  const vector = new Float64Array(DIM)
  vector[0] = 1
  return vector
})()

/** 关键词 → 向量：in-<i> / out-<j> / 其它（查询用）。 */
const embedding: ExperienceEmbeddingPort = {
  source: "local",
  dimension: DIM,
  async embed(texts: string[]): Promise<Float64Array[]> {
    return texts.map((text) => {
      const inMatch = /in-(\d+)/u.exec(text)
      if (inMatch) return inVector(Number(inMatch[1]))
      const outMatch = /out-(\d+)/u.exec(text)
      if (outMatch) return outVector(Number(outMatch[1]))
      return UNKNOWN
    })
  },
}

function runtime(): ExperienceRuntimePort {
  return {
    activeRunForSession: () => null,
    runForChild: () => null,
    hasTeamInCurrentRun: () => false,
    hasActiveRun: () => false,
    modelForCaller: () => "verifier-model",
  }
}

function portOf(store: AssetStore): ExperienceStorePort {
  return {
    nextId: () => store.nextId(),
    getActivePrompt: (type) => store.getActivePrompt(type),
    listPrompts: () => store.listPrompts(),
    listRows: (limit) => store.listRows(limit),
    getRows: (ids, options) => store.getRows(ids, options),
    listActiveEmbeddings: (type) => store.listActiveExperienceEmbeddings(type),
    insertChecked: (input) => store.insertChecked(input),
    updateFields: (id, patch, next) => store.updateFields(id, patch, next),
    setActive: (id, active) => store.setActive(id, active),
    recordUsage: (input) => store.recordUsage(input),
    listInjectedIds: (input) => store.listInjectedIds(input),
    insertEvaluationsChecked: (input) => store.insertEvaluationsChecked(input),
    getStats: (ids) => store.getStats(ids),
    rebuildStats: (input) => store.rebuildStats(input),
  }
}

async function makeStore(now: () => number): Promise<AssetStore> {
  const dir = await mkdtemp(join(tmpdir(), "vw-verify-pool-"))
  cleanups.push(() => rm(dir, { recursive: true, force: true }))
  const store = new AssetStore(dir, { now })
  await store.init()
  cleanups.push(async () => store.close())
  return store
}

const caller = { isChild: false, sessionId: "verify-pool" }

function candidate(keyword: string): Record<string, unknown> {
  return {
    responsibility: `${keyword} 的责任归属`,
    task_type: "软件开发",
    decision_domain: `${keyword} 的决策域`,
    situation: `${keyword} 情形下需要判断前置条件`,
    trigger: `准备处理 ${keyword} 时`,
    principle: `${keyword} 原则：前置条件未稳定时并行会放大返工`,
    recommended_action: `${keyword} 行动：先建立完成闸门`,
    exclusions: [`${keyword} 不适用时`],
    evidence: [`一次 ${keyword} 相关执行`],
  }
}

function perfect(experienceId: string) {
  return { experienceId, fitScore: 1 as const, decisionEffect: 1 as const, informationGain: 1 as const, causalConfidence: 1 as const, evidence: "" }
}

describe("§36 Trust 只能在 semantic Top-K 内重排", () => {
  it("test_池外高信任经验_修正后相关性更高也不得进入结果", async () => {
    const store = await makeStore(() => 5_000)
    const service = new ExperienceService({
      store: portOf(store),
      runtime: runtime(),
      embedding,
      now: () => 5_000,
      logger: { warn: () => undefined },
    })
    await service.initializePrompt({ caller, type: "agent" })

    const inKeywords = Array.from({ length: CANDIDATE_POOL_SIZE }, (_unused, index) => `in-${index}`)
    const outKeywords = Array.from({ length: 5 }, (_unused, index) => `out-${index}`)
    for (const batch of [inKeywords.slice(0, 8), inKeywords.slice(8, 16), inKeywords.slice(16, 24), inKeywords.slice(24), outKeywords]) {
      const result = await service.submit({ caller, type: "agent", candidates: { experiences: batch.map(candidate) } })
      expect(result.inserted).toHaveLength(batch.length)
    }

    const outIds: string[] = []
    const all = await store.listRows(100)
    for (const entry of all) {
      if (entry.taskRetrievalText.includes("out-")) outIds.push(entry.id)
    }
    expect(outIds).toHaveLength(5)

    // 池外 5 条：各 16 次满分评价 → qualitySignal ≈ 0.7281（有效样本量偏离中性）
    for (const id of outIds) {
      for (let index = 0; index < 16; index += 1) {
        await service.recall({ caller, type: "agent", ids: [id] })
        const feedback = await service.feedback({ caller, type: "agent", evaluations: [perfect(id)] })
        expect(feedback.accepted).toHaveLength(1)
      }
    }
    const outStats = (await store.getStats([outIds[0]]))[0] as ExperienceStatsEntry
    // eslint-disable-next-line no-console
    console.log("DEBUG outIds=", JSON.stringify(outIds), "stats=", JSON.stringify(await store.getStats(outIds)))
    expect(outStats.usedCount).toBe(16)
    expect(outStats.qualitySignal).toBeGreaterThan(0.7)

    // 手算：池外经验的修正后相关性（0.745 × (1 + 0.2 × q)）严格高于池内经验的 0.75
    const outAdjusted = calculateRecallAdjustment({ cosine: OUT_COS, qualitySignal: outStats.qualitySignal })
    const inAdjusted = calculateRecallAdjustment({ cosine: IN_COS, qualitySignal: 0 })
    expect(outAdjusted).toBeGreaterThan(inAdjusted)
    expect(outAdjusted).toBeGreaterThan(0.85)

    const recalled = await service.recall({ caller, type: "agent", query: "查询", topK: 50 })
    if (recalled.kind !== "candidates") throw new Error("应为候选摘要")
    // 语义候选池上限 = 30：池外经验一律不得进入有界重排
    expect(recalled.hits).toHaveLength(CANDIDATE_POOL_SIZE)
    expect(recalled.hits.every((hit) => !outIds.includes(hit.id))).toBe(true)
    expect(recalled.hits.every((hit) => hit.score === inAdjusted)).toBe(true)
    // eslint-disable-next-line no-console
    console.log(`\n池边界证据：池内 adjusted=${inAdjusted} 池外 adjusted=${outAdjusted.toFixed(6)} 返回条数=${recalled.hits.length}\n`)
  })
})

describe("§24 使用计数增量不刷新 updated_at", () => {
  it("test_推进时钟下_第二次使用只涨计数且updated_at不变", async () => {
    let clock = 1_000
    const store = await makeStore(() => {
      clock += 1_000
      return clock
    })
    const service = new ExperienceService({
      store: portOf(store),
      runtime: runtime(),
      embedding: {
        source: "local",
        dimension: 2,
        async embed(texts: string[]): Promise<Float64Array[]> {
          return texts.map(() => new Float64Array([1, 0]))
        },
      },
      now: () => clock,
      logger: { warn: () => undefined },
    })
    await service.initializePrompt({ caller, type: "agent" })
    const submitted = await service.submit({ caller, type: "agent", candidates: { experiences: [candidate("唯一")] } })
    const id = submitted.inserted[0].id

    await service.recall({ caller, type: "agent", ids: [id] })
    const first = (await store.getStats([id]))[0] as ExperienceStatsEntry
    expect(first.recalledCount).toBe(1)
    const clockAfterFirst = clock

    await service.recall({ caller, type: "agent", ids: [id] })
    const second = (await store.getStats([id]))[0] as ExperienceStatsEntry
    expect(second.recalledCount).toBe(2)
    // 推进时钟已前进，但 updated_at 未变 → 增量路径确实没有刷新记账时间
    expect(clock).toBeGreaterThan(clockAfterFirst)
    expect(second.updatedAt).toBe(first.updatedAt)

    await service.feedback({ caller, type: "agent", evaluations: [perfect(id)] })
    const third = (await store.getStats([id]))[0] as ExperienceStatsEntry
    expect(third.updatedAt).toBeGreaterThan(first.updatedAt)
    // eslint-disable-next-line no-console
    console.log(`\n时钟证据：usage#1 updated_at=${first.updatedAt} usage#2 updated_at=${second.updatedAt} clock=${clock} feedback 后 updated_at=${third.updatedAt}\n`)
  })
})
