// 独立验证者临时文件（task-8 复核轮 D1）：
//   1. 20 条锚点定义真的进入**注册后的工具 description**（模型可见面），且逐条、互不重复。
//   2. 运行时扫描整个 src/，确认每条定义文本在**唯一一处**本体里出现（无第二份定义）。
//   3. 用 §5 的语义关键词做抽样对照，确认英文定义确实对应方案里的行为锚点（不是相对措辞）。
// 验证完成后删除。

import { describe, expect, it } from "vitest"
import { readFile, readdir } from "node:fs/promises"
import { fileURLToPath } from "node:url"
import { join } from "node:path"
import {
  EVALUATION_DIMENSION_DEFINITIONS,
  renderAnchorGlossary,
} from "../../src/host/experience/index.js"
import { registerWfExperienceFeedback } from "../../src/host/tools/wf-experience-feedback/tool.js"

/** 捕获注册结果的最小 ctx（工具注册只经 ctx.get('tools').register）。 */
function captureDefinition(): { description: string; parameters: Record<string, unknown> } {
  const captured = new Map<string, { description: string; parameters: Record<string, unknown> }>()
  const ctx = {
    get(name: string): unknown {
      if (name !== "tools") return undefined
      return {
        register(definition: { name: string } & Record<string, unknown>): () => void {
          captured.set(definition.name, definition as never)
          return () => undefined
        },
      }
    },
  }
  registerWfExperienceFeedback(ctx, {} as never)
  const definition = captured.get("wf_experience_feedback")
  if (!definition) throw new Error("工具未注册")
  return definition
}

/** 递归收集 src/ 下全部 ts/tsx 文件正文。 */
async function collectSources(): Promise<Array<{ path: string; text: string }>> {
  const root = fileURLToPath(new URL("../../src/", import.meta.url))
  const collected: Array<{ path: string; text: string }> = []
  const walk = async (dir: string): Promise<void> => {
    for (const entry of await readdir(dir, { withFileTypes: true })) {
      const full = join(dir, entry.name)
      if (entry.isDirectory()) {
        await walk(full)
        continue
      }
      if (!entry.name.endsWith(".ts") && !entry.name.endsWith(".tsx")) continue
      collected.push({ path: full.replace(root, "").replace(/\\/gu, "/"), text: await readFile(full, "utf8") })
    }
  }
  await walk(root)
  return collected
}

/** §5 锚点语义的关键词对照（抽样：每维至少 2 级）。 */
const DOC_ALIGNMENT: ReadonlyArray<{ dimension: string; value: number; keywords: readonly string[] }> = [
  { dimension: "fitScore", value: 0, keywords: ["key conditions", "does not apply"] },
  { dimension: "fitScore", value: 1, keywords: ["transfers directly"] },
  { dimension: "decisionEffect", value: -1, keywords: ["wrong way", "risk", "rework", "loss"] },
  { dimension: "decisionEffect", value: 0, keywords: ["no attributable difference"] },
  { dimension: "informationGain", value: 0, keywords: ["common sense", "no extra decision information"] },
  { dimension: "informationGain", value: 1, keywords: ["non-obvious", "discriminating"] },
  { dimension: "causalConfidence", value: 0, keywords: ["cannot tell whether the experience caused"] },
  { dimension: "causalConfidence", value: 1, keywords: ["counterfactual"] },
]

describe("D1 复核：锚点定义进入模型可见面", () => {
  it("test_注册后的工具description_含四维20条逐级定义且各出现一次", () => {
    const definition = captureDefinition()
    let checked = 0
    for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
      expect(definition.description, `缺参数名 ${dimension.parameter}`).toContain(dimension.parameter)
      expect(definition.description, `缺问题 ${dimension.dimension}`).toContain(dimension.question)
      for (const anchor of dimension.anchors) {
        const entry = `${anchor.value} = ${anchor.definition}`
        expect(definition.description, `缺锚点 ${dimension.dimension}@${anchor.value}`).toContain(entry)
        expect(definition.description.split(entry).length - 1, `锚点重复 ${entry}`).toBe(1)
        checked += 1
      }
    }
    expect(checked).toBe(20)
    expect(definition.description).toContain(renderAnchorGlossary())
    // eslint-disable-next-line no-console
    console.log(`\nD1 证据：description 长度=${definition.description.length}，逐级定义命中=${checked}/20\n`)
  })

  it("test_20条定义_互不重复且不是相对措辞(每维5级行为描述彼此可区分)", () => {
    const definitions: string[] = []
    for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
      expect(dimension.anchors).toHaveLength(5)
      for (const anchor of dimension.anchors) {
        expect(anchor.definition.split(/\s+/u).length, `${dimension.dimension}@${anchor.value} 过短`).toBeGreaterThanOrEqual(4)
        // 相对措辞（只有较低/中等/较高）不构成行为锚点
        expect(anchor.definition.toLowerCase()).not.toMatch(/^(lower|higher|medium|moderate|average)\.?$/u)
        definitions.push(anchor.definition)
      }
    }
    expect(new Set(definitions).size).toBe(20)
  })

  it("test_逐级定义文本_在src全域只出现在锚点定义本体一处", async () => {
    const sources = await collectSources()
    expect(sources.length).toBeGreaterThan(100)
    for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
      for (const anchor of dimension.anchors) {
        const holders = sources.filter((file) => file.text.includes(anchor.definition)).map((file) => file.path)
        expect(holders, `锚点定义 ${anchor.definition} 出现于 ${holders.join(",")}`).toEqual([
          "host/experience/anchor-definitions.ts",
        ])
      }
    }
  })

  it("test_英文定义与方案§5行为锚点_关键词抽样一致", () => {
    for (const item of DOC_ALIGNMENT) {
      const dimension = EVALUATION_DIMENSION_DEFINITIONS.find((entry) => entry.dimension === item.dimension)
      if (!dimension) throw new Error(`缺维度 ${item.dimension}`)
      const anchor = dimension.anchors.find((entry) => entry.value === item.value)
      if (!anchor) throw new Error(`缺锚点 ${item.dimension}@${item.value}`)
      for (const keyword of item.keywords) {
        expect(anchor.definition.toLowerCase(), `${item.dimension}@${item.value} 缺语义 ${keyword}`)
          .toContain(keyword.toLowerCase())
      }
    }
  })

  it("test_参数级描述_不复制逐级定义文本", () => {
    const definition = captureDefinition()
    const properties = (definition.parameters as { properties?: Record<string, { description?: string }> }).properties
    expect(properties).toBeTruthy()
    for (const [name, property] of Object.entries(properties ?? {})) {
      const text = property.description ?? ""
      for (const dimension of EVALUATION_DIMENSION_DEFINITIONS) {
        for (const anchor of dimension.anchors) {
          expect(text.includes(anchor.definition), `${name} 复制了锚点定义`).toBe(false)
        }
      }
    }
  })
})
